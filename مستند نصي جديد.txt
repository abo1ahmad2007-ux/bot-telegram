// 🏢 المحرك الإداري لأدمن الكليات (Sub-Admin Panel)

function handleSubAdminLogic(chatId, text, state, facultyDetails) {

config.deleteProperty(chatId);
var menu = {
keyboard: [
[{text: "📥 صندوق شكاوى الكلية"}, {text: "📢 إعلانات الكلية الموجهة"}],
[{text: "👥 إدارة طلاب الكلية"}, {text: "🎛️ تخصيص محلي للأزرار"}],
[{text: "🔙 لوحة الكلية"}]
],
resize_keyboard: true
};
sendText(chatId, "🏢 مرحباً بك في لوحة تحكم الكلية والأقسام الفرعية\nالقسم المسؤول عنه: " + facultyDetails.faculty + " \nصلاحيتك الممنوحة: " + facultyDetails.role + " ", menu);
return;
}

if (text === "📥 صندوق شكاوى الكلية") {
if (facultyDetails.role === "مشرف كلية - إعلانات") {
sendText(chatId, "❌ حسابك يمتلك صلاحية نشر الإعلانات فقط، لا يمكنك إدارة صندوق الشكاوى.");
return;
}
sendSubAdminTickets(chatId, facultyDetails.faculty);
return;
}

if (text === "📢 إعلانات الكلية الموجهة") {
if (facultyDetails.role === "مشرف كلية - شكاوى") {
sendText(chatId, "❌ حسابك يمتلك صلاحية إدارة الشكاوى فقط، لا يمكنك نشر الإعلانات.");
return;
}
config.setProperty(chatId, "SUB_WAIT_BROADCAST");
sendText(chatId, "📢 أرسل نص الإعلان المراد توجيهه فقط وحصرياً لطلاب كلية (" + facultyDetails.faculty + "):", {keyboard: [[{text: "🔙 لوحة الكلية"}]], resize_keyboard: true});
return;
}

if (text === "👥 إدارة طلاب الكلية") {
sendText(chatId, getFacultyStats(facultyDetails.faculty));
return;
}
}

// ⚙️ نظام معالجة المدخلات والحالات العالمي (States Engine)
// ========================================================
function handleGlobalStateEngine(chatId, text, state, facultyDetails) {

// معالجة إرسال شكوى من طالب
if (state === "WAIT_MSG") {
var userData = getUserInfo(chatId);
recordTicket(chatId, text, userData);
return;
}

// الرد على شكوى من مشرف
if (state.startsWith("REPLY_TO_TICKET_")) {
var parts = state.split("_");
var studentChatId = parts[3];
var ticketRow = parts[4];

sendText(studentChatId, "✅ وصلك رد رسمي من ممثلي الهيئة الطلابية بخصوص تذكرتك:\n\n💬 " + text);

var sheet = SpreadsheetApp.openById(ssId).getSheetByName("الشكاوى");
sheet.getRange(ticketRow, 10).setValue("تمت المعالجة");
sheet.getRange(ticketRow, 12).setValue(text);

config.deleteProperty(chatId);
sendText(chatId, "✅ تم إرسال الرد وتحديث حالة الشكوى بنجاح.");
logAdminAction(chatId, "الرد على الشكوى رقم " + (ticketRow-1) + " وإغلاقها", facultyDetails ? facultyDetails.faculty : "الإدارة العليا");
return;
}

// التواصل المباشر مع طالب
if (state === "SADMIN_WAIT_DIRECT_ID") {
config.setProperty(chatId + "_DIR_ID", text);
config.setProperty(chatId, "SADMIN_WAIT_DIRECT_MSG");
sendText(chatId, "✍️ اكتب الرسالة أو التنبيه المراد توجيهه مباشرةً للطالب الآن:");
return;
}
if (state === "SADMIN_WAIT_DIRECT_MSG") {
var targetId = config.getProperty(chatId + "_DIR_ID");
sendText(targetId, "💬 رسالة مباشرة من إدارة الهيئة الطلابية العليا:\n\n" + text);
config.deleteProperty(chatId); config.deleteProperty(chatId + "_DIR_ID");
sendText(chatId, "✅ تم إرسال الرسالة بنجاح وبسرية تامة.");
return;
}

// جدولة إعلانات
if (state === "SADMIN_WAIT_SCHED_MSG") {
config.setProperty(chatId + "_SCHED_TXT", text);
config.setProperty(chatId, "SADMIN_WAIT_SCHED_TIME");
sendText(chatId, "🕒 الآن حدد وقت وتاريخ النشر بالصيغة التالية تماماً:\nyyyy-MM-dd HH:mm\nمثال: 2026-06-15 14:30");
return;
}
if (state === "SADMIN_WAIT_SCHED_TIME") {
var sTxt = config.getProperty(chatId + "_SCHED_TXT");
var sheet = SpreadsheetApp.openById(ssId).getSheetByName("الجدولة والنسخ الاحتياطي");
sheet.appendRow([new Date(), "إعلان مجدول", sTxt, text, "قيد الانتظار"]);
config.deleteProperty(chatId); config.deleteProperty(chatId + "_SCHED_TXT");
sendText(chatId, "✅ تم حفظ وجدولة الإعلان بنجاح في قاعدة البيانات وسيتم النشر تلقائياً بالوقت المحدد.");
return;
}

// الحظر وإساءة الاستخدام
if (state === "SADMIN_WAIT_BAN_ID") {
var targetBanId = text;
var sheet = SpreadsheetApp.openById(ssId).getSheetByName("احصائيات الدخول");
var data = sheet.getDataRange().getValues();
var found = false;
for (var i = 1; i < data.length; i++) {
if (data[i][5].toString().trim() === targetBanId) {
var currentStatus = data[i][8] || "";
var newStatus = currentStatus === "محظور 🚫" ? "نشط ✅" : "محظور 🚫";
sheet.getRange(i + 1, i === 1 ? 9 : 9).setValue(newStatus); // تعيين حالة الحظر في العمود التاسع
sendText(chatId, "🔄 تم تغيير حالة حساب الطالب (" + data[i][2] + ") إلى: " + newStatus + " ");
found = true; break;
}
}
if (!found) sendText(chatId, "❌ لم يتم العثور على طالب يحمل هذا الـ Chat ID.");
config.deleteProperty(chatId);
return;
}

// إعلانات الكلية الفرعية
if (state === "SUB_WAIT_BROADCAST") {
config.deleteProperty(chatId);
sendText(chatId, "⏳ جاري تعميم الإعلان وتوجيهه لكافة طلاب قسمك بالخلفية...");
broadcastToFacultyStudents(facultyDetails.faculty, text);
sendText(chatId, "✅ تم نشر الإعلان وتوصيله بنجاح لطلاب كليتك فقط.");
logAdminAction(chatId, "نشر إعلان موجه ومخصص للكلية", facultyDetails.faculty);
return;
}

// التسجيل للمرة الأولى (بيانات الطالب)
if (state === "WAIT_NAME") {
config.setProperty(chatId + "_N", text); config.setProperty(chatId, "WAIT_PHONE");
var phoneKeyboard = { keyboard: [[{ text: "📱 مشاركة رقم الهاتف الرسمي", request_contact: true }]], resize_keyboard: true, one_time_keyboard: true };
sendText(chatId, "✅ تم حفظ الاسم.\nالآن يرجى الضغط على الزر بالأسفل لمشاركة رقم هاتفك المربوط تليجرام تلقائياً للتوثيق والتحقق البشري:", phoneKeyboard);
return;
}
if (state === "WAIT_PHONE") {
if (data.message.contact) {
config.setProperty(chatId + "_P", data.message.contact.phone_number);
config.setProperty(chatId, "WAIT_UNI");
sendText(chatId, "✅ تم توثيق رقم الهاتف بنجاح.\nأرسل الآن رقمك الجامعي الرسمي (مثال: 20261012):", {remove_keyboard: true});
} else {
sendText(chatId, "⚠️ خطأ! يرجى الضغط على الزر المخصص بالأسفل حصراً لمشاركة رقم الهاتف الحقيقي للتوثيق.");
}
return;
}
if (state === "WAIT_UNI") {
config.setProperty(chatId + "_U", text); config.setProperty(chatId, "WAIT_YEAR");
var yearMenu = { keyboard: [[{text:"سنة أولى"}, {text:"سنة ثانية"}], [{text:"سنة ثالثة"}, {text:"سنة رابعة"}], [{text:"سنة خامسة"}]], resize_keyboard: true };
sendText(chatId, "🔢 اختر أو اكتب السنة الدراسية الحالية:", yearMenu);
return;
}
if (state === "WAIT_YEAR") {
config.setProperty(chatId + "_Y", text); config.setProperty(chatId, "WAIT_FACULTY");
sendFacultyKeyboard(chatId, "🎓 أخيراً، اختر كليتك من الأزرار بالأسفل لإتمام بناء حسابك الموحد:");
return;
}

// حاسبة المعدل GPA
if (state === "WAIT_GPA_CALC") {
config.deleteProperty(chatId);
calculateAndSendGPA(chatId, text);
return;
}

// خطوط النقل
if (state === "WAIT_BUS_CITY") {
config.deleteProperty(chatId);
sendBusLineDetails(chatId, text);
return;
}

// 🧮 المحرك الحسابي المتقدم للمعدلات والأكاديميات (GPA Calculator)
// ========================================================
function calculateAndSendGPA(chatId, rawText) {
try {
var lines = rawText.split("\n");
var totalPoints = 0;
var totalHours = 0;
var details = "📈 تفصيل درجات ومعدل المواد:\n\n";

for (var i = 0; i < lines.length; i++) {
if (!lines[i].includes(":")) continue;
var parts = lines[i].split(":");
var grade = parseFloat(parts[0].trim());
var hours = parseFloat(parts[1].trim());

if (isNaN(grade) || isNaN(hours)) continue;

// تحويل نظام العلامات المئوية في AUST إلى نقاط النقاط الأكاديمية (4.0 Scale)
var points = 0;
var statusStr = "ناجح ✅";
if (grade >= 90) points = 4.0;
else if (grade >= 85) points = 3.5;
else if (grade >= 80) points = 3.0;
else if (grade >= 75) points = 2.5;
else if (grade >= 70) points = 2.0;
else if (grade >= 60) points = 1.5;
else if (grade >= 50) points = 1.0;
else { points = 0.0; statusStr = "راسب ❌"; }

totalPoints += (points * hours);
totalHours += hours;
details += "🔹 مادة بواقع (" + hours + " س): علامة " + grade + " ➡️ نقاط: " + points + " [" + statusStr + "]\n";
}

if (totalHours === 0) {
sendText(chatId, "❌ لم يتم إدخال البيانات بصيغة صحيحة. يرجى مراجعة المثال الموضح وإعادة المحاولة.");
return;
}

var finalGPA = (totalPoints / totalHours).toFixed(2);
details += "\n=========================\n🧮 المعدل الفصلي النهائي الناتج: " + finalGPA + " / 4.00";

sendText(chatId, details);
} catch(e) {
sendText(chatId, "❌ حدث خطأ في معالجة العمليات الحسابية للمعدل.");
}
}
